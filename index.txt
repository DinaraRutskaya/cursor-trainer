<script>
    const maze = document.getElementById('maze');
    const msg = document.getElementById('msg');
    let active = false;

    document.getElementById('start').onmouseenter = () => {
        active = true;
        msg.innerText = "Вези по белой дорожке...";
        msg.style.color = "black";
    };

    maze.onmouseenter = (e) => {
        if (e.target === maze && active) {
            active = false;
            msg.innerText = "Ошибка! Вернись на Старт.";
            msg.style.color = "red";
        }
    };

    document.getElementById('finish').onmouseenter = () => {
        if (active) {
            active = false;
            msg.innerText = "🎉 Упражнение пройдено!";
            msg.style.color = "green";
        }
    };
</script>
